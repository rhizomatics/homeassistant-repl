1"""The client-side half of `sql` results: a separate, independent SqlResult 

2from custom_components/ha_repl_server/sql.py's, sharing nothing but the 

3Arrow IPC bytes `.arrow()` produces there and `SqlResult.from_arrow()` 

4reads here - see this project's docs/developer/design/design_principles.md 

5("maximum freedom for the developer on the client, maximum restrictions on 

6the server side"). The server's own job stops at running the query and 

7downloading a complete (`max_rows`-capped, not streamed - HA's own tables 

8are small enough that lazy/paged loading isn't worth the complexity yet) 

9in-memory Arrow dataset; everything past that - rendering (`.show()`), 

10dataframe conversion (`.to_pandas()`/`.to_polars()`), CSV export - runs 

11here, against that already-downloaded data, so installing pandas/polars 

12(or anything else) is a `pip install` in your own environment, never a 

13change to the running Home Assistant instance. 

14 

15Once downloaded, a SqlResult behaves like a wrapped local database handle 

16in its own right (think `sqlite3`'s cursor, except the storage underneath 

17is an Arrow buffer, not a `.db` file) - every operation below (slicing, 

18`.project()`, `.sample()`, iteration, ...) works off that one in-memory 

19copy, no further round trip to Home Assistant. 

20"""

190 @classmethod 

191 def from_arrow( 

192 cls, 

193 data: bytes, 

194 *, 

195 truncated: bool = False, 

196 tables: list[SqlTable] | None = None, 

197 legacy: bool = False, 

198 hidden: frozenset[str] = frozenset(), 

199 query: str = "", 

200 ) -> SqlResult: 

201 """Rebuild a SqlResult from Arrow IPC stream bytes (the one shared 

202 contract with the server that produced them) - a single 

203 struct-typed batch, one named field per column. `truncated` isn't 

204 itself encoded in the Arrow data (it's metadata about the query, 

205 not the rows), so it travels alongside the bytes rather than 

206 inside them. `hidden` names the columns left out unless `legacy`, 

207 and `query` is the query the bytes are the result of. 

208 """ 

209 batch = na.ArrayStream.from_readable(data).read_all() 

210 names = {field.name for field in batch.schema.fields} 

211 matches = [t for t in tables or () if names <= {c.name for c in t._columns}] 

212 table = matches[0] if len(matches) == 1 else None 

213 return cls(batch, truncated, table, legacy, hidden, query)

215 @classmethod

218 ) -> SqlResult: 

219 """A result over these per-column arrays, sharing their buffers.""" 

220 schema = na.struct({name: arr.schema for name, arr in columns.items()})

222 schema, length=rowcount, buffers=[], children=list(columns.values()) 

223 ) 

224 return dataclasses.replace(like, _data=na.Array(batch))

243 def arrow_ipc(self) -> bytes: 

244 """The result serialized as an Arrow IPC stream - the format 

245 `from_arrow()` reads, and what `polars.read_ipc_stream()` or 

246 `pyarrow.ipc.open_stream()` expect, e.g. to save to a file. 

247 """ 

248 shown = self._shown() 

249 data = shown._data 

250 if data.n_chunks == 1 and data.offset: 

251 # The IPC writer can't encode a sliced (offset) array; write a 

252 # compact copy of just these rows instead. 

253 data = shown._take(range(self.rowcount))._data 

254 buf = io.BytesIO() 

255 with StreamWriter.from_writable(buf) as writer: 

256 writer.write_stream(data) 

257 return buf.getvalue()

310 def show( 

311 self, 

312 columns: list[str] | None = None, 

313 *, 

314 max_rows: int | None = DEFAULT_SHOW_ROWS, 

315 max_cols: int | None = None, 

316 ) -> Table: 

317 """A rich Table rendering of this result. Shows the same columns 

318 a row of it has (`result[0]`), unless `columns` names the ones to 

319 show or `max_cols` caps how many; at most `max_rows` rows. 

320 """ 

321 all_names = self.column_names 

322 if columns is not None: 

323 names = columns 

324 cols_truncated = False 

325 elif max_cols is not None and len(all_names) > max_cols: 

326 names = all_names[:max_cols] 

327 cols_truncated = True 

328 else: 

329 names = all_names 

330 cols_truncated = False 

331 

332 if max_rows is not None and self.rowcount > max_rows: 

333 row_cap = max_rows 

334 rows_truncated = True 

335 else: 

336 row_cap = self.rowcount 

337 rows_truncated = False

344 if rows_truncated: 

345 caption += f" (showing first {row_cap} rows)" 

346 

347 table = Table(*names, caption=caption) 

348 for i, row in enumerate(self._rows(names)): 

349 if i >= row_cap: 

350 break 

351 table.add_row(*("" if v is None else str(v) for v in row)) 

352 return table

354 def project(self, columns: list[str]) -> SqlResult: 

355 """A new SqlResult with just these columns (same rows, same 

356 underlying arrays - no data is copied). A legacy column named 

357 here is in view in the new result. 

358 """ 

359 arrays = self._all_arrays() 

360 projected = self._from_columns( 

361 {name: arrays[name] for name in columns}, self.rowcount, self 

362 ) 

363 projected._hidden = self._hidden - set(columns) 

364 return projected

366 def __getitem__(self, key: int | slice) -> Any: 

367 """One row by position (`r[0]`, `r[-1]`) as a SqlRow, or a 

368 contiguous run of rows with standard Python slice notation 

369 (`r[:10]`, `r[-1:]`, `r[10:20]`) - a new SqlResult sharing this 

370 one's data, nothing copied. For anything more (a step, a filter, 

371 a sort) use a dataframe: `r.to_polars()`. 

372 """ 

373 if isinstance(key, int) and not isinstance(key, bool): 

374 index = key + self.rowcount if key < 0 else key 

375 if not 0 <= index < self.rowcount: 

376 raise IndexError( 

377 f"row {key} out of range - result has {self.rowcount} " 

378 f"row{'' if self.rowcount == 1 else 's'}" 

379 ) 

380 arrays = self._column_arrays() 

381 return SqlRow( 

382 [column[index].as_py() for column in arrays.values()], 

383 list(arrays), 

384 self.table, 

385 ) 

386 if not isinstance(key, slice): 

387 raise TypeError( 

388 "SqlResult takes a row number (result[0]) or a slice " 

389 f"(result[:10]), not {key!r}" 

390 ) 

391 start, stop, step = key.indices(self.rowcount) 

392 if step != 1: 

393 raise ValueError( 

394 "SqlResult slices can't have a step - use result.to_polars() " 

395 "for anything beyond a contiguous run of rows" 

396 ) 

397 if self._data.n_chunks != 1: # nothing to share: an empty result 

398 return self 

399 sliced = na.c_array(self._data)[start : max(start, stop)] 

400 return dataclasses.replace(self, _data=na.Array(sliced))

414 def _take(self, indices: Any) -> SqlResult: 

415 """A new SqlResult holding copies of just these rows, in this order 

416 - only the rows asked for are read, not the whole column.""" 

417 columns = { 

418 name: na.array([_stored(column[i].as_py()) for i in indices], column.schema) 

419 for name, column in self._all_arrays().items() 

420 }

425 not Home Assistant's) - a header row, then every row with each

447 def _all_arrays(self) -> dict[str, Any]: 

448 """As _column_arrays(), whether or not the column is in view.""" 

449 children = list(self._data.iter_children()) 

450 if self._data.n_chunks == 1 and ( 

451 self._data.offset or any(len(c) != self.rowcount for c in children) 

452 ): 

453 # A sliced struct keeps its offset/length on itself; its child 

454 # arrays still span the original rows, so narrow them to match. 

455 start, stop = self._data.offset, self._data.offset + self.rowcount 

456 children = [na.Array(na.c_array(c)[start:stop]) for c in children] 

457 names = [field.name for field in self._data.schema.fields] 

458 return dict(zip(names, children, strict=True))

508@dataclass 

509class SqlTool: 

510 """Live mode's `sql`: a local object, not a name on the Home Assistant 

511 side. Calling it sends the query over the websocket, downloads the 

512 whole (`max_rows`-capped) result as Arrow and hands back a local 

513 SqlResult - so `sql(...).show()`, `.to_polars()` and anything else you 

514 do with the result runs in this process, with whatever you've 

515 installed here. 

516 

517 `.max_rows` is this shell's default row cap (set it to change the 

518 default for every call after, or to None to remove it entirely) and 

519 `.tables` the recorder's tables, real and virtual, to explore the 

520 schema without a query 

521 - `.table(name)` picks one of them out by name. 

522 """ 

523 

524 client: Any 

525 max_rows: int | None = DEFAULT_MAX_ROWS 

526 tables: list[SqlTable] = field(default_factory=list) 

527 

528 @classmethod 

529 async def connect(cls, client: Any) -> SqlTool: 

530 """A SqlTool with `.tables` already downloaded. Raises HaReplError 

531 if the server has no sql to offer (switched off in the 

532 integration's options, or a server too old to have the command).""" 

533 reply = await client.call("ha_repl_server/sql_tables") 

534 tables: list[SqlTable] = [] 

535 for t in reply["tables"]: 

536 columns = tuple( 

537 SqlColumn(c["name"], c["type"], c.get("legacy", False)) 

538 for c in t["columns"] 

539 ) 

540 # class and doc are absent from a server older than this client 

541 described = (t["name"], columns, t.get("class", ""), t.get("doc", "")) 

542 if t.get("virtual"): 

543 tables.append( 

544 SqlVirtualTable(*described, definition=t.get("definition", "")) 

545 ) 

546 else: 

547 tables.append(SqlTable(*described)) 

548 return cls(client, tables=tables)

550 def table(self, name: str, *, legacy: bool = False) -> SqlTable: 

551 """The recorder table called `name`, from `.tables` - with its 

552 legacy columns too if `legacy`. Raises KeyError, naming the 

553 tables there are, if there's no such table.""" 

554 for table in self.tables: 

555 if table.name == name: 

556 return dataclasses.replace(table, legacy=True) if legacy else table 

557 known = ", ".join(t.name for t in self.tables) 

558 raise KeyError(f"no table {name!r} - tables are: {known}")

560 async def __call__( 

561 self, 

562 query: str, 

563 *, 

564 max_rows: int | None = _MAX_ROWS_UNSET, 

565 legacy: bool = False, 

566 raw: bool = False, 

567 ) -> SqlResult: 

568 """Run a single read-only SELECT against Home Assistant's Recorder 

569 database. At most `max_rows` rows are fetched (default: this 

570 object's own `.max_rows`) - pass None for no cap at all. 

571 

572 A recorder column comes back as what it holds: a `_ts` or 

573 datetime column as a datetime in UTC, a context id as its text. 

574 With `raw` they are left as the database has them - seconds 

575 since the epoch as a float, the id's bytes. 

576 

577 Legacy columns - still in a table, no longer written to - are 

578 downloaded but kept out of view unless the query names them or 

579 `legacy` is True, so a `select *` shows only the columns in use. 

580 Setting `.legacy = True` on the result brings them into view.""" 

581 params: dict[str, Any] = { 

582 "query": query, 

583 "max_rows": self.max_rows if max_rows is _MAX_ROWS_UNSET else max_rows, 

584 } 

585 if raw: # left out otherwise, which an older server needs 

586 params["raw"] = True 

587 try: 

588 reply = await self.client.call("ha_repl_server/sql", **params) 

589 except HaReplError as err: 

590 if err.code != "sql_error": 

591 raise 

592 raise SqlError(err.detail) from None 

593 return SqlResult.from_arrow( 

594 base64.b64decode(reply["arrow"]),

602 def _tables_of(self, query: str) -> list[SqlTable]: 

603 """The tables a result of this query could be described by: the 

604 virtual tables it mentions - going by its words - or, if none, the 

605 real ones. A virtual table's columns are also those of the tables 

606 it joins, so without this neither could be told from the other.""" 

607 words = set(re.findall(r"\w+", query.lower())) 

608 virtual = [t for t in self.tables if t.virtual and t.name.lower() in words] 

609 return virtual or [t for t in self.tables if not t.virtual] 

610 

611 def _unnamed_legacy(self, query: str) -> frozenset[str]: 

612 """Names of legacy columns of the tables this query mentions that 

613 the query doesn't itself mention - going by its words, not a 

614 parse. A name that's also a column in use in another of those 

615 tables (`entity_id`, in a join of states and states_meta) isn't 

616 one of them.""" 

617 words = set(re.findall(r"\w+", query.lower())) 

618 tables = [t for t in self.tables if t.name.lower() in words] 

619 in_use = {c.name for t in tables for c in t._columns if not c.legacy} 

620 legacy = {c.name for t in tables for c in t._columns if c.legacy} 

621 return frozenset(n for n in legacy - in_use if n.lower() not in words)
