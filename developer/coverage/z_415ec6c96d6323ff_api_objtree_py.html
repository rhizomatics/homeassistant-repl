1"""ApiObjTree - API client mode's `obj[...]`, built entirely from 

2Home Assistant's standard websocket API (get_states, config/*_registry/list) 

3- no ha_repl_server component needed, works against any instance an admin 

4token can reach. Bound to the same name, `obj`, as the live tree (see 

5apirepl.py) so a snippet that only touches `obj` runs unchanged in either 

6mode. Entities only have what that API exposes: state + attributes from 

7get_states, and whatever the entity registry's own partial dict exposes (no 

8live component instance, so no calling methods on it, no fields an 

9integration keeps off the registry/state). 

10 

11Actions (services) are in the tree too, from get_services, under the 

12integration that registers them: `obj["/mqtt/action/publish"]`. Looking one 

13up reads the snapshot like anything else here; calling it is the one thing 

14in this tree that isn't read-only, and goes to Home Assistant as a 

15call_service. 

16 

17Mirrors custom_components/ha_repl_server/objtree.py's shape - __getitem__, 

18.keys()/.values()/.items(), find(), show() all behave the same way modulo the 

19smaller attribute set - but lives in this package rather than that one (a 

20HACS-deployed component with its own packaging boundary), so the handful of 

21small, stable, HA-import-free pieces it needs (the ordered-view mixin, 

22parse_path, the None/empty-collection cleanup) are duplicated here rather than 

23imported across that boundary. Revisit if API client mode sticks and a shared 

24package becomes worth the packaging work. 

25 

26Unlike the live tree, this one is explicitly a *snapshot*: fetching is async 

27(it's a websocket call) but Mapping's `__getitem__`/`__iter__`/`__len__` are 

28not, so refreshing can't happen lazily on access without risking asyncio 

29reentrancy (this is normally used from inside an already-running event loop). 

30Instead the owning REPL loop decides when to call `await cache.refresh()` 

31(e.g. between prompts, when `cache.is_stale()`) - see apirepl.py. `reset_cache()` 

32just marks the cache stale; it does not fetch. 

33"""

551def _clean(data: dict[str, Any]) -> dict[str, Any]: 

552 """Recursively drop None/empty-collection values and render the known 

553 timestamp fields as local ISO 8601 strings. 

554 

555 Empty *lists* are dropped here too, not just empty sets: labels/aliases 

556 etc. arrive over the wire as JSON lists (JSON has no set type), so a list 

557 is this mode's wire-equivalent of the live tree's empty-set fields - and 

558 keeping both modes behaving the same way is the point. 

559 """ 

560 cleaned: dict[str, Any] = {} 

561 for key, value in data.items(): 

562 if key in _TIMESTAMP_FIELDS and isinstance(value, (int, float)): 

563 value = datetime.fromtimestamp(value).astimezone().isoformat() 

564 else: 

565 value = _normalize(value) 

566 if value is None or value == set() or value == []: 

567 continue 

568 cleaned[key] = value 

569 return cleaned

572@dataclass(frozen=True) 

573class ApiObjTree(Mapping[str, "ApiEntity | ApiAction | ApiObjTree"]): 

574 """The tree of Home Assistant's entities and actions, as read through its API. 

575 

576 Entities are arranged by integration, then domain, then name: 

577 

578 obj["/mqtt"] everything from one integration 

579 obj["/mqtt/sensor"] its sensors 

580 obj["/mqtt/sensor/shed_temp"] one entity 

581 obj["sensor.shed_temp"] the same entity, by entity id 

582 

583 A missing path or entity raises KeyError. Each level is a dict of the 

584 level below, in alphabetical order: len(), `in`, keys(), values() and 

585 items() all work on it, and those three can be indexed like a list. 

586 

587 find() goes through every entity below a path instead, in no fixed 

588 order. It takes a path or regular expression, and any of domain=, 

589 platform=, area=, label= and object_type=, each a string or a list 

590 of strings. 

591 find_paths() and find_names() take the same and give tree paths or 

592 entity ids rather than the entities. 

593 

594 show(path) gives the most useful details of one entity. 

595 

596 An integration's actions (services) are beside its domains: 

597 

598 obj["/mqtt/action"] its actions 

599 obj["/mqtt/action/publish"] one action - call it to run it 

600 

601 The find methods go through actions as well as entities. Give them 

602 object_type="entity" or object_type="action" for just the one kind. 

603 

604 This is a snapshot, refreshed between prompts once it is a while old. 

605 reset_cache() makes the next prompt fetch a new one. 

606 """

707 def _find( 

708 self, 

709 path: str, 

710 *, 

711 platform: str | list[str] | None, 

712 domain: str | list[str] | None, 

713 area: str | list[str] | None, 

714 label: str | list[str] | None, 

715 object_type: str | list[str] | None = None, 

716 ) -> Iterator[_Found]: 

717 """The real search, shared by find()/find_paths()/find_names() - each 

718 just projects a different field from the (path, entity) pairs this 

719 yields. 

720 

721 `path` is normally an exact /integration/domain/object_id prefix, the 

722 same as indexing - but a path containing a regex metacharacter (e.g. 

723 "/mqtt/binary_sensor/barn.*") is matched as a regular expression 

724 against each candidate's full path instead, anchored at the start 

725 (so it behaves like a prefix match unless you anchor the end 

726 yourself with `$`). That trades the usual early narrowing for a scan 

727 of this view's whole subtree, filtered by the pattern. 

728 

729 `domain` matches the HA domain (e.g. "light") across 

730 integrations, the same way `platform` does for the owning 

731 integration; `area`/`label` each take an id or a display name. Each 

732 of the four ORs within itself when given a list, and they AND 

733 together. No order guarantee; wrap in sorted(...) if you want one. 

734 

735 Actions are found along with entities, each with "action" as its 

736 domain and the integration whose action it is as its platform. An 

737 action has no area or label, so a search by either finds none. 

738 

739 `object_type` narrows it to "entity", "action" or "device", or a 

740 list of those - all of them when it isn't given. 

741 """ 

742 _require_str(path) 

743 scope = tuple(p for p in (self.integration, self.domain) if p is not None) 

744 pattern = None 

745 if path not in ("", "/") and _REGEX_METACHARS.intersection(path): 

746 pattern = re.compile(path) 

747 prefix = scope 

748 else: 

749 try: 

750 prefix = scope if path in ("", "/") else scope + parse_path(path) 

751 except ValueError as err: 

752 raise KeyError(str(err)) from None 

753 if len(prefix) > 3: 753 ↛ 754line 753 didn't jump to line 754 because the condition on line 753 was never true

754 raise KeyError(path) 

755 

756 platforms = _as_set(platform) 

757 domains = _as_set(domain) 

758 types = _object_types(object_type) 

759 area_ids = _resolve_from_cache(area, self.cache.areas, "area", "area_id") 

760 label_ids = _resolve_from_cache(label, self.cache.labels, "label", "label_id") 

761 

762 def _matches() -> Iterator[_Found]: 

763 for entity in self._entries(prefix) if "entity" in types else (): 

764 if platforms and entity.platform not in platforms: 

765 continue 

766 if domains and entity.domain not in domains: 

767 continue 

768 if area_ids and entity.area_id not in area_ids: 

769 continue 

770 if label_ids and not (entity.labels & label_ids): 

771 continue 

772 full_path = "/" + "/".join( 

773 (entity.platform, entity.domain, entity.object_id)[len(scope) :] 

774 ) 

775 if pattern is not None and not pattern.match(full_path): 

776 continue 

777 yield _Found(full_path, entity) 

778 if area_ids or label_ids or "action" not in types: 

779 return 

780 for action in self._actions(prefix): 

781 if platforms and action.domain not in platforms: 

782 continue 

783 if domains and ACTION not in domains: 

784 continue 

785 full_path = "/" + "/".join( 

786 (action.domain, ACTION, action.service)[len(scope) :] 

787 ) 

788 if pattern is not None and not pattern.match(full_path): 

789 continue 

790 yield _Found(full_path, action) 

791 

792 return _matches()
