132async def _maybe_await(value: Any) -> Any: 

133 """Finish a coroutine (or other awaitable) the user forgot to `await`; 

134 anything else passes straight through. Used both by the AST rewrite below 

135 (an unawaited call nested inside a larger expression) and _run_code's own 

136 check (a bare reference to one created earlier, e.g. `c = f(); c`).""" 

137 return await value if inspect.isawaitable(value) else value

140def unawait(value: Any) -> Any: 

141 """Identity function and auto-await escape hatch: `unawait(f())` returns 

142 f()'s bare, un-awaited result (a coroutine, if f is async) - for when 

143 that's genuinely wanted, e.g. batching into `asyncio.gather(*[unawait(f()) 

144 for f in ...])`. Recognised by name in the `_AutoAwait` rewrite below, 

145 which skips its whole argument rather than calling this at runtime; this 

146 plain version only runs if auto-await itself is off (--no-auto-await) or 

147 `unawait` is used somewhere the rewrite doesn't reach. 

148 """ 

149 return value

152class _AutoAwait(ast.NodeTransformer): 

153 """Rewrites every call not already explicitly awaited to go through 

154 _maybe_await() first, so `obj.async_method()` works whether or not the 

155 user remembered `await` - including nested inside attribute access 

156 (`obj.async_method().attr`), a comprehension, or an argument list. 

157 Leaves nested (synchronous) function/lambda bodies alone: `await` there 

158 is a SyntaxError, and those calls run later, not as part of this 

159 statement anyway. `unawait(expr)` is the escape hatch - its argument is 

160 left completely untouched for when the bare coroutine is wanted. 

161 """

181 def visit_Await(self, node: ast.Await) -> ast.Await: 

182 # Already explicit: don't double-await this call, but still rewrite 

183 # anything nested inside its own arguments.

185 node.value.func = self.visit(node.value.func) 

186 node.value.args = [self.visit(a) for a in node.value.args] 

187 node.value.keywords = [self.visit(k) for k in node.value.keywords] 

188 else:

207async def _run_code(code: Any, globals_: dict[str, Any]) -> Any: 

208 # No separate "bare coroutine" backstop here: when auto_await is on, the 

209 # _AutoAwait rewrite already resolves every call site, and NOT doing so 

210 # unconditionally is exactly what `unawait(...)` asks for - a backstop 

211 # here would silently defeat it for a trailing `unawait(f())`. 

212 result = eval(code, globals_) # nosec B307 - the whole point of a dev shell 

213 if code.co_flags & inspect.CO_COROUTINE: 

214 result = await result 

215 return result

218def _user_traceback(err: BaseException) -> Any: 

219 # Drop the shell's own leading frames so the traceback starts at user 

220 # code (compiled under /ha_repl/, see _execute) - whichever module of 

221 # ours called in. An error with no user frame at all is left whole. 

222 tb = err.__traceback__ 

223 while tb is not None and not tb.tb_frame.f_code.co_filename.startswith("/ha_repl/"): 

224 tb = tb.tb_next 

225 return tb or err.__traceback__
